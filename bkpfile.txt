        function validateCheckout() {
                let name = document.getElementById("customer-name").value.trim();

                checkoutMessage.classList.remove("hidden");

    function loadCart() {
        let cart = JSON.parse(localStorage.getItem("cart")) || [];
        let cartContainer = document.getElementById("cart-items");
        let totalElement = document.querySelector(".cart-total");
        let emptyCartMessage = document.querySelector(".empty-cart");
        let chkfrm = document.querySelector(".checkout-form");

        cartContainer.innerHTML = "";
        let totalAmount = 0;

        if (cart.length === 0) {
            emptyCartMessage.style.display = "block";
            totalElement.style.display = "none";
            chkfrm.style.display = "none";
            return;
        } else {
            emptyCartMessage.style.display = "none";
            totalElement.style.display = "block";
        }

        cart.forEach(item => {
            let totalItemPrice = item.price * item.quantity;
            totalAmount += totalItemPrice;

            let cartItem = `
                <div class="cart-item">
                    <div class="cart-item-details">
                        <img src="${item.image}" alt="${item.name}">
                        <div class="cart-item-info">
                            <h3>${item.name}</h3>
                            <p>Price: ₹${item.price}</p>
                            <p>Total: ₹${totalItemPrice}</p>
                        </div>
                    </div>
                    <div class="quantity">
                        <button onclick="updateQuantity(${item.id}, -1)">-</button>
                        <input type="text" value="${item.quantity}" readonly>
                        <button onclick="updateQuantity(${item.id}, 1)">+</button>
                    </div>
                    <button class="btn-remove" onclick="removeItem(${item.id})">❌ Remove</button>
                </div>
            `;
            cartContainer.innerHTML += cartItem;
        });

        totalElement.innerHTML = `Total: ₹${totalAmount}`;
    }

    function updateQuantity(id, change) {
        let cart = JSON.parse(localStorage.getItem("cart")) || [];
        let itemIndex = cart.findIndex(item => item.id == id);

        if (itemIndex !== -1) {
            cart[itemIndex].quantity += change;
            if (cart[itemIndex].quantity <= 0) {
                cart.splice(itemIndex, 1);
            }
        }

        localStorage.setItem("cart", JSON.stringify(cart));
        loadCart();
    }

    function removeItem(id) {
        let cart = JSON.parse(localStorage.getItem("cart")) || [];
        cart = cart.filter(item => item.id != id);

        localStorage.setItem("cart", JSON.stringify(cart));
        loadCart();
    }

    document.querySelector(".clear-cart").addEventListener("click", function () {
        localStorage.removeItem("cart");
        loadCart();
    });

    loadCart();
